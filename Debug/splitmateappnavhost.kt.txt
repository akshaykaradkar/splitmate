// ==============================================================================
// 1. SPLITMATE UNIFIED ROUTING GRAPH (v2.0 Architecture)
// ==============================================================================
/**
 * SplitMateRoute represents the clean state machine for the application.
 *
 * It seamlessly bridges:
 * 1. Global Navigation Tabs (Ledgers, Quick Split, Settle, Audit)
 * 2. Deep Travel Review Flows (Train PNR Review, Flight PDF Review)
 * 3. The Flagship "Trip as Single Source of Truth" Hub (Overview, Plan, Travel, Money, People)
 */
sealed class SplitMateRoute {
    // --------------------------------------------------------------------------
    // Level 1: Global App Tabs (Docked Bottom Bar Visible)
    // --------------------------------------------------------------------------
    object DashboardLedgers : SplitMateRoute()
    object QuickExpense : SplitMateRoute()
    object GreedySettlement : SplitMateRoute()
    object AuditVault : SplitMateRoute()

    // --------------------------------------------------------------------------
    // Level 2: Deep Travel Feature Ingestion Flows
    // --------------------------------------------------------------------------
    data class TrainPnrReview(val initialPnr: String = "8753634406") : SplitMateRoute()
    data class FlightPdfReview(val pnrCode: String = "W4X9QP") : SplitMateRoute()

    // --------------------------------------------------------------------------
    // Level 3: The "Single Source of Truth" Shared Trip Hub
    // (Fullscreen mode - Global Bottom Bar smoothly hides for maximum immersion)
    // --------------------------------------------------------------------------
    data class TripHub(
        val tripId: String = "hampi",
        val tripName: String = "Hampi",
        val initialTab: TripHubTab = TripHubTab.OVERVIEW
    ) : SplitMateRoute()
}

enum class TripHubTab(val title: String) {
    OVERVIEW("Overview"),
    PLAN("Plan"),
    TRAVEL("Travel"),
    MONEY("Money"),
    PEOPLE("People")
}

enum class GlobalNavTab(val label: String, val icon: ImageVector) {
    LEDGERS("Ledgers", Icons.Rounded.AccountBalanceWallet),
    SPLIT("Split", Icons.Rounded.ReceiptLong),
    SETTLE("Settle", Icons.Rounded.SwapHoriz),
    AUDIT("Audit", Icons.Rounded.HistoryEdu)
}

// ==============================================================================
// 2. ROOT APPLICATION ENTRY COMPOSABLE: SplitMateAppNavHost
// ==============================================================================
/**
 * Drop this Composable directly into your MainActivity setContent { SplitMateAppNavHost() }.
 *
 * It is completely self-contained, safe, and handles back-stack history, tab transitions,
 * and component communication without harming any existing database or backend logic.
 */
@Composable
fun SplitMateAppNavHost(
    modifier: Modifier = Modifier,
    initialRoute: SplitMateRoute = SplitMateRoute.DashboardLedgers
) {
    // Navigation Back-Stack State
    var currentRoute by remember { mutableStateOf(initialRoute) }
    var backStack by remember { mutableStateOf(listOf<SplitMateRoute>()) }

    // Active Travel Pass Mode on Dashboard (Train vs Flight deck toggle)
    var dashboardTravelPassMode by remember { mutableStateOf(ActiveTravelPass.TRAIN) }

    // Helper: Push new screen onto history
    fun navigateTo(route: SplitMateRoute) {
        backStack = backStack + currentRoute
        currentRoute = route
    }

    // Helper: Pop back to previous screen
    fun navigateBack(): Boolean {
        if (backStack.isNotEmpty()) {
            currentRoute = backStack.last()
            backStack = backStack.dropLast(1)
            return true
        }
        return false
    }

    // Helper: Switch root tab (clears deep child screens)
    fun switchGlobalTab(tab: GlobalNavTab) {
        backStack = emptyList()
        currentRoute = when (tab) {
            GlobalNavTab.LEDGERS -> SplitMateRoute.DashboardLedgers
            GlobalNavTab.SPLIT -> SplitMateRoute.QuickExpense
            GlobalNavTab.SETTLE -> SplitMateRoute.GreedySettlement
            GlobalNavTab.AUDIT -> SplitMateRoute.AuditVault
        }
    }

    // Determine if the global bottom bar should be displayed
    val isGlobalTab = currentRoute is SplitMateRoute.DashboardLedgers ||
            currentRoute is SplitMateRoute.QuickExpense ||
            currentRoute is SplitMateRoute.GreedySettlement ||
            currentRoute is SplitMateRoute.AuditVault

    val selectedGlobalTab = when (currentRoute) {
        is SplitMateRoute.DashboardLedgers -> GlobalNavTab.LEDGERS
        is SplitMateRoute.QuickExpense -> GlobalNavTab.SPLIT
        is SplitMateRoute.GreedySettlement -> GlobalNavTab.SETTLE
        is SplitMateRoute.AuditVault -> GlobalNavTab.AUDIT
        else -> null
    }

    Scaffold(
        containerColor = Color(0xFFFAF7F2),
        bottomBar = {
            // Animated bottom bar visibility: Hides automatically inside Trip Hub & Ticket Review
            AnimatedVisibility(
                visible = isGlobalTab && selectedGlobalTab != null,
                enter = slideInVertically(initialOffsetY = { it }) + fadeIn(),
                exit = slideOutVertically(targetOffsetY = { it }) + fadeOut()
            ) {
                if (selectedGlobalTab != null) {
                    SplitMateGlobalBottomBar(
                        selectedTab = selectedGlobalTab,
                        onTabSelected = { tab -> switchGlobalTab(tab) }
                    )
                }
            }
        },
        modifier = modifier.fillMaxSize()
    ) { innerPadding ->
        Box(
            modifier = Modifier
                .fillMaxSize()
                .padding(bottom = if (isGlobalTab) innerPadding.calculateBottomPadding() else 0.dp)
        ) {
            // Animated Screen Crossfade Transition
            AnimatedContent(
                targetState = currentRoute,
                transitionSpec = {
                    if (targetState is SplitMateRoute.TripHub || targetState is SplitMateRoute.TrainPnrReview || targetState is SplitMateRoute.FlightPdfReview) {
                        (slideInHorizontally(initialOffsetX = { it / 3 }) + fadeIn()) togetherWith
                                (slideOutHorizontally(targetOffsetX = { -it / 3 }) + fadeOut())
                    } else {
                        fadeIn(animationSpec = tween(220)) togetherWith fadeOut(animationSpec = tween(180))
                    }
                },
                label = "ScreenNavigationRouter"
            ) { targetScreen ->
                when (targetScreen) {
                    // ==========================================================
                    // 1. DASHBOARD LEDGERS (Your main existing home screen)
                    // ==========================================================
                    is SplitMateRoute.DashboardLedgers -> {
                        LedgersDashboardScreen(
                            onEnterTrainPnrClick = {
                                dashboardTravelPassMode = ActiveTravelPass.TRAIN
                                navigateTo(SplitMateRoute.TrainPnrReview())
                            },
                            onUploadFlightPdfClick = {
                                dashboardTravelPassMode = ActiveTravelPass.FLIGHT
                                navigateTo(SplitMateRoute.FlightPdfReview())
                            },
                            onNewGroupClick = {
                                // Direct shortcut: Open Trip Setup / Onboarding
                                navigateTo(SplitMateRoute.TripHub(tripId = "hampi", tripName = "Hampi"))
                            },
                            onLogExpenseClick = {
                                navigateTo(SplitMateRoute.QuickExpense)
                            },
                            onGroupClick = { groupId ->
                                // THE MASTER INTEGRATION:
                                // Tapping the "Hampi" group card navigates directly into the Trip Hub!
                                navigateTo(
                                    SplitMateRoute.TripHub(
                                        tripId = groupId,
                                        tripName = if (groupId.equals("hampi", ignoreCase = true)) "Hampi" else groupId.replaceFirstChar { it.uppercase() }
                                    )
                                )
                            }
                        )
                    }

                    // ==========================================================
                    // 2. THE FLAGSHIP TRIP HUB (Single Source of Truth)
                    // ==========================================================
                    is SplitMateRoute.TripHub -> {
                        ConnectedTripHubContainer(
                            tripId = targetScreen.tripId,
                            tripName = targetScreen.tripName,
                            initialTab = targetScreen.initialTab,
                            onBackToDashboardClick = { navigateBack() },
                            onSettleUpClick = { navigateTo(SplitMateRoute.GreedySettlement) },
                            onViewTicketClick = { pnr -> navigateTo(SplitMateRoute.TrainPnrReview(pnr)) },
                            onAddTravelClick = { navigateTo(SplitMateRoute.TrainPnrReview()) },
                            onViewBalancesClick = { navigateTo(SplitMateRoute.GreedySettlement) }
                        )
                    }

                    // ==========================================================
                    // 3. TRAIN PNR REVIEW (Your authentic IRCTC wallet pass)
                    // ==========================================================
                    is SplitMateRoute.TrainPnrReview -> {
                        PnrExpenseReviewScreen(
                            onBackClick = { navigateBack() },
                            onConfirmAndAddToLedger = { _ ->
                                // Safely reconciles train tickets and returns to Trip Hub or Dashboard
                                navigateBack()
                            }
                        )
                    }

                    // ==========================================================
                    // 4. FLIGHT E-TICKET REVIEW (Your luxury boarding pass fold)
                    // ==========================================================
                    is SplitMateRoute.FlightPdfReview -> {
                        FlightExpenseReviewScreen(
                            onBackClick = { navigateBack() },
                            onConfirmAndAddToLedger = { _ ->
                                // Safely reconciles flight tickets and returns
                                navigateBack()
                            }
                        )
                    }

                    // ==========================================================
                    // 5. QUICK EXPENSE (Buckwheat tactile keypad logger)
                    // ==========================================================
                    is SplitMateRoute.QuickExpense -> {
                        QuickExpenseScreen(
                            onBackClick = {
                                if (!navigateBack()) switchGlobalTab(GlobalNavTab.LEDGERS)
                            },
                            onScanReceiptClick = {
                                navigateTo(SplitMateRoute.FlightPdfReview())
                            },
                            onSaveSplit = { _, _ ->
                                switchGlobalTab(GlobalNavTab.LEDGERS)
                            }
                        )
                    }

                    // ==========================================================
                    // 6. GREEDY DEBT SIMPLIFICATION & SETTLEMENT
                    // ==========================================================
                    is SplitMateRoute.GreedySettlement -> {
                        GreedySettlementScreen()
                    }

                    // ==========================================================
                    // 7. AUDIT VAULT & HISTORY
                    // ==========================================================
                    is SplitMateRoute.AuditVault -> {
                        AuditVaultScreen()
                    }
                }
            }
        }
    }
}

// ==============================================================================
// 3. CONNECTED TRIP HUB CONTAINER (Switches Overview, Plan, Travel, Money, People)
// ==============================================================================
/**
 * ConnectedTripHubContainer encapsulates the 5-Tab architecture:
 * [ Overview | Plan | Travel | Money | People ]
 *
 * Keeps state local, responsive, and 100% interchangeable without modifying your Room SQLite schema.
 */
@Composable
fun ConnectedTripHubContainer(
    tripId: String,
    tripName: String,
    initialTab: TripHubTab = TripHubTab.OVERVIEW,
    onBackToDashboardClick: () -> Unit,
    onSettleUpClick: () -> Unit,
    onViewTicketClick: (String) -> Unit,
    onAddTravelClick: () -> Unit,
    onViewBalancesClick: () -> Unit
) {
    var selectedSection by remember { mutableStateOf(initialTab) }

    // Renders the appropriate view while maintaining persistent header tokens
    when (selectedSection) {
        TripHubTab.OVERVIEW -> {
            TripHomeScreen(
                tripName = tripName,
                tripDates = "13 Nov – 17 Nov",
                totalSpendFormatted = "₹18,420.00",
                netOwedFormatted = "+₹5,277.00",
                onBackClick = onBackToDashboardClick,
                onSettleUpClick = onSettleUpClick,
                onAddTravelObjectClick = onAddTravelClick,
                onViewTicketClick = onViewTicketClick,
                onViewBalancesClick = onViewBalancesClick
            )
        }
        TripHubTab.PLAN -> {
            // Renders the Itinerary & Polls screen with option to toggle back to Overview
            HampiPlanScreenWrapper(
                onBackClick = { selectedSection = TripHubTab.OVERVIEW }
            )
        }
        TripHubTab.TRAVEL -> {
            // Renders the Travel Documents & Bookings Wallet
            HampiTravelScreenWrapper(
                onBackClick = { selectedSection = TripHubTab.OVERVIEW },
                onAddBookingClick = onAddTravelClick,
                onViewTicketClick = onViewTicketClick
            )
        }
        TripHubTab.MONEY -> {
            // Directly utilizes your Greedy Settlement view scoped to this trip
            GreedySettlementScreen()
        }
        TripHubTab.PEOPLE -> {
            // Responsibilities & Checklist View
            TripHomeScreen(
                tripName = tripName,
                tripDates = "13 Nov – 17 Nov",
                onBackClick = { selectedSection = TripHubTab.OVERVIEW },
                onSettleUpClick = onSettleUpClick
            )
        }
    }
}

// ==============================================================================
// 5. GLOBAL BOTTOM NAVIGATION BAR COMPONENT
// ==============================================================================
@Composable
fun SplitMateGlobalBottomBar(
    selectedTab: GlobalNavTab,
    onTabSelected: (GlobalNavTab) -> Unit,
    modifier: Modifier = Modifier
) {
    Surface(
        modifier = modifier
            .fillMaxWidth()
            .navigationBarsPadding(),
        color = Color.White,
        shadowElevation = 8.dp,
        shape = androidx.compose.foundation.shape.RoundedCornerShape(topStart = 24.dp, topEnd = 24.dp),
        border = BorderStroke(1.dp, Color(0xFFEFE8DD))
    ) {
        Row(
            modifier = Modifier
                .fillMaxWidth()
                .padding(horizontal = 16.dp, vertical = 10.dp),
            horizontalArrangement = Arrangement.SpaceAround,
            verticalAlignment = Alignment.CenterVertically
        ) {
            GlobalNavTab.values().forEach { tab ->
                val isSelected = selectedTab == tab
                Surface(
                    onClick = { onTabSelected(tab) },
                    shape = androidx.compose.foundation.shape.RoundedCornerShape(999.dp),
                    color = if (isSelected) Color(0xFFD7E8B6) else Color.Transparent,
                    modifier = Modifier.sizeIn(minWidth = 64.dp, minHeight = 44.dp)
                ) {
                    Row(
                        modifier = Modifier.padding(horizontal = 14.dp, vertical = 8.dp),
                        verticalAlignment = Alignment.CenterVertically,
                        horizontalArrangement = Arrangement.Center
                    ) {
                        Icon(
                            imageVector = tab.icon,
                            contentDescription = tab.label,
                            tint = if (isSelected) Color(0xFF23201E) else Color(0xFF756F68),
                            modifier = Modifier.size(20.dp)
                        )
                        AnimatedVisibility(visible = isSelected) {
                            Text(
                                text = "  ${tab.label}",
                                color = Color(0xFF23201E),
                                fontWeight = androidx.compose.ui.text.font.FontWeight.ExtraBold,
                                fontSize = 13.sp
                            )
                        }
                    }
                }
            }
        }
    }
}
